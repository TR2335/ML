## 写计划阶段的裁定（Plan Rulings）

1. **R1 — outcomes 用 dict 行而非新领域类型**：`derive_labels(..., outcomes: list[dict], ...)`，每行 `{encounter_id, occurred_at, outcome_type}`（`occurred_at` 接受 ISO 字符串或 datetime，naive 按 UTC）。不动 serving 的 `event_type` Literal（新增 RRT 等类型会扩大 API 契约，超出 P5 范围）。代价：管线侧少一层类型检查，由 `OUTCOME_TYPES` 白名单兜底。
2. **R2 — `LabeledLandmark` 定义在 `scripts/derive_labels.py`**：`dataclass(frozen=True)`，字段 `encounter_id/index_time/label`。app 层不依赖管线产物类型。
3. **R3 — demo 阳性率的确定性设计**：18 患者 × index 小时网格 1..96（每人 96 行 → 共 1728 行）；高风险 4 人（{2,5,9,14}）RRT 发生在首事件 +48h；阳性 index ∈ (24h, 48] → 每人 24 行 → 96/1728 = **5.56%**，稳定落在 [5%, 15%]。
4. **R4 — 评估报告结构向后兼容**：新增 `report["comparison"] = {model, news2, model_with_news2}` 与 `report["news2_feature_gain"]`；`report["evaluation"]` 保留为 model 指标（既有消费方不破坏）。
5. **R5 — T5.4 触及 `train_model.py`**：WBS 明文「此变更必须同步写入 T5.3 的工件契约 `feature_names`」——训练第 11 特征 `news2_score`（`TARGET_CONFIG` 增 news2 生成权重：deterioration 0.06、readmission 0.02，作用于原始 NEWS2 分 0–15）。文件清单以修法文字为准。
6. **R6 — 特征窗口为半开区间** `[index - window_hours, index)`：上界严格早于 index（防泄漏，WBS 测试明文），下界闭（含边界测量）。
7. **R7 — `build_features` CLI 以标签 CSV 为锚**：按 `(encounter_id, index_time)` 字符串精确合并；`--demo` 用 seed=7 重建与 T5.1 同构的事件。
8. **R8 — 因子标签回退**：`_score` 构造 `RiskFactor` 时用 `LABELS.get(name, name)`——新特征名（七元组/`news2_score`）无中文标签时回退原名，避免 KeyError。
9. **R9 — NEWS2 缺项策略**：五个组件（RR/SpO2/SBP/HR/体温）缺测则跳过该项累加；意识项始终缺测（demo 无 ACVPU），报告与文档注明局限；任一组件得 3 分 → 直接 high（官方升级规则）；评估用归一化 `score/15`，high 阈值 `7/15`。
10. **R10 — 中位数填补的特征仍计入 `missing_features`**：`preprocess` 路径下缺失特征用 median 填补参与计算，但同时列入 `missing_features`（可行动性诚实，解释因子显示填补值）。
11. **R11 — `preprocess` 必须伴随 `feature_order`**：旧形状工件（无 `feature_order`）携带 `preprocess` 视为非法（`ValueError`），避免静默忽略部署方以为生效的标准化配置。
12. **R12 — labeled_csv 分支暂不支持子群组分析**：`--input-format labeled_csv` 同时传 `--group-column` 时直接拒绝（`SystemExit`），默认格式行为不变；子群组留给真实数据阶段。

```python
DEMO_SERIAL_HOURS = range(0, 97, 6)  # 每 6h 一条生命体征，0..96h
DEMO_RRT_DELAY_HOURS = 48


def demo_patients_with_outcomes(
    count: int = 18, seed: int = 7
) -> tuple[list[dict], list[ClinicalEvent], list[dict]]:
    """演示管线专用：患者 + 串行生命体征 + 高风险患者的 RRT 结局（与真实管道同构）。"""
    patients, first_events = demo_patients(count=count, seed=seed)
    events: list[ClinicalEvent] = []
    outcomes: list[dict] = []
    for patient, first in zip(patients, first_events):
        encounter_id = patient["encounter_id"]
        index = int(encounter_id.split("-")[1])
        for hour in DEMO_SERIAL_HOURS:
            events.append(
                first.model_copy(
                    update={
                        "event_id": f"{first.event_id}-h{hour}",
                        "occurred_at": first.occurred_at + timedelta(hours=hour),
                    }
                )
            )
        if index in DEMO_HIGH_RISK:
            outcomes.append(
                {
                    "encounter_id": encounter_id,
                    "occurred_at": (
                        first.occurred_at + timedelta(hours=DEMO_RRT_DELAY_HOURS)
                    ).isoformat(),
                    "outcome_type": "rrt",
                }
            )
    return patients, events, outcomes
```

```python
"""Derive landmark labels from outcome events (T5.1).

Demo-first CLI: real-data entry stays the importable derive_labels() function
until the P9 self-service data path lands.
"""

import argparse
import csv
import json
from dataclasses import dataclass
from datetime import UTC, datetime, timedelta
from pathlib import Path

from app.domain import ClinicalEvent
from app.synthetic import FEATURES, demo_patients_with_outcomes

OUTCOME_TYPES = frozenset({"rrt", "unplanned_icu_transfer", "cardiac_arrest", "death"})
DEMO_INDEX_HOURS = range(1, 97)


@dataclass(frozen=True)
class LabeledLandmark:
    encounter_id: str
    index_time: datetime
    label: int


def _as_datetime(value: datetime | str) -> datetime:
    parsed = value if isinstance(value, datetime) else datetime.fromisoformat(str(value))
    if parsed.tzinfo is None:
        parsed = parsed.replace(tzinfo=UTC)
    return parsed.astimezone(UTC)


def derive_labels(
    events: list[ClinicalEvent],
    outcomes: list[dict],
    index_times: list[tuple[str, datetime]],
    horizon_hours: int = 24,
) -> list[LabeledLandmark]:
    """Label each (encounter, index) landmark: 1 when a whitelisted outcome
    falls in [index, index + horizon), else 0. Encounters must appear in events."""
    encounters = {event.encounter_id for event in events}
    horizon = timedelta(hours=horizon_hours)
    by_encounter: dict[str, list[datetime]] = {}
    for outcome in outcomes:
        if str(outcome.get("outcome_type", "")) not in OUTCOME_TYPES:
            continue
        encounter_id = str(outcome["encounter_id"])
        by_encounter.setdefault(encounter_id, []).append(_as_datetime(outcome["occurred_at"]))
    labeled: list[LabeledLandmark] = []
    for encounter_id, raw_index in sorted(index_times):
        if encounter_id not in encounters:
            continue
        index_time = _as_datetime(raw_index)
        end = index_time + horizon
        label = int(
            any(
                index_time <= occurred < end
                for occurred in by_encounter.get(encounter_id, [])
            )
        )
        labeled.append(
            LabeledLandmark(encounter_id=encounter_id, index_time=index_time, label=label)
        )
    return labeled


def demo_labeled_landmarks() -> list[LabeledLandmark]:
    _, events, outcomes = demo_patients_with_outcomes()
    first_seen: dict[str, datetime] = {}
    for event in sorted(events, key=lambda item: item.occurred_at):
        first_seen.setdefault(event.encounter_id, event.occurred_at)
    index_times = {
        encounter_id: first + timedelta(hours=hour)
        for encounter_id, first in first_seen.items()
        for hour in DEMO_INDEX_HOURS
    }
    return derive_labels(events, outcomes, index_times)


def _latest_payload_before(
    events: list[ClinicalEvent], encounter_id: str, index_time: datetime
) -> dict[str, object]:
    candidates = [
        event
        for event in events
        if event.encounter_id == encounter_id and event.occurred_at < index_time
    ]
    if not candidates:
        return {}
    latest = max(candidates, key=lambda event: event.occurred_at)
    return {
        feature: latest.payload[feature]
        for feature in FEATURES
        if latest.payload.get(feature) is not None
    }


def main() -> None:
    parser = argparse.ArgumentParser(description="Derive landmark labels (demo mode in P5)")
    parser.add_argument("--demo", action="store_true", help="use synthetic patients/outcomes")
    parser.add_argument("--output", default="data/demo_labels.csv")
    parser.add_argument("--horizon-hours", type=int, default=24)
    args = parser.parse_args()
    if not args.demo:
        raise SystemExit("P5 仅实现 --demo；真实数据请调用 derive_labels()（见 docs/data-contract.md）")
    _, events, outcomes = demo_patients_with_outcomes()
    first_seen: dict[str, datetime] = {}
    for event in sorted(events, key=lambda item: item.occurred_at):
        first_seen.setdefault(event.encounter_id, event.occurred_at)
    index_times = {
        encounter_id: first + timedelta(hours=hour)
        for encounter_id, first in first_seen.items()
        for hour in DEMO_INDEX_HOURS
    }
    labeled = derive_labels(events, outcomes, index_times, horizon_hours=args.horizon_hours)
    output = Path(args.output)
    output.parent.mkdir(parents=True, exist_ok=True)
    with output.open("w", newline="", encoding="utf-8") as handle:
        writer = csv.writer(handle)
        writer.writerow(["encounter_id", "index_time", "label", *FEATURES])
        for item in labeled:
            payload = _latest_payload_before(events, item.encounter_id, item.index_time)
            writer.writerow(
                [
                    item.encounter_id,
                    item.index_time.isoformat(),
                    item.label,
                    *[payload.get(feature, "") for feature in FEATURES],
                ]
            )
    positives = sum(item.label for item in labeled)
    print(
        json.dumps(
            {
                "rows": len(labeled),
                "positives": positives,
                "positive_rate": round(positives / len(labeled), 4),
                "output": str(output),
            },
            ensure_ascii=False,
        )
    )


if __name__ == "__main__":
    main()
```

```python
"""Shared feature order for training and serving (T5.2+).

The seven-tuple expansion (latest/mean/slope/min/max/missing/since_last) is
the contract both the offline pipeline and future trainers consume; serving
keeps raw-feature scoring until artifacts carry their own feature_order.
"""

from datetime import datetime, timedelta

from app.domain import ClinicalEvent
from app.synthetic import FEATURES

STAT_SUFFIXES = ("latest", "mean", "slope", "min", "max", "missing", "since_last")

TUPLE_FEATURE_ORDER: list[str] = [
    f"{feature}__{suffix}" for feature in FEATURES for suffix in STAT_SUFFIXES
]

# T5.4 在七元组之后追加 news2_score（stacking 基线输入）。
FEATURE_ORDER: list[str] = list(TUPLE_FEATURE_ORDER)


def build_landmarks(
    patient_events: dict[str, list[ClinicalEvent]],
    index_times: list[tuple[str, datetime]],
    window_hours: int = 24,
) -> list[dict]:
    """Build seven-tuple features per (encounter, index) landmark.

    Window is half-open [index - window_hours, index): events at the index
    instant are future information and must never leak in (Plan Ruling R6).
    index_times is a list of (encounter_id, index_time) pairs (Plan Ruling R13).
    """
    window = timedelta(hours=window_hours)
    rows: list[dict] = []
    for encounter_id, index_time in sorted(index_times):
        window_events = sorted(
            (
                event
                for event in patient_events.get(encounter_id, [])
                if index_time - window <= event.occurred_at < index_time
            ),
            key=lambda event: event.occurred_at,
        )
        row: dict = {"encounter_id": encounter_id, "index_time": index_time.isoformat()}
        for feature in FEATURES:
            samples = [
                (event.occurred_at, float(event.payload[feature]))
                for event in window_events
                if event.payload.get(feature) is not None
            ]
            row.update(_feature_stats(feature, samples, index_time))
        rows.append(row)
    return rows


def _feature_stats(
    feature: str, samples: list[tuple[datetime, float]], index_time: datetime
) -> dict[str, float | None]:
    prefix = f"{feature}__"
    if not samples:
        return {
            prefix + "latest": None,
            prefix + "mean": None,
            prefix + "slope": None,
            prefix + "min": None,
            prefix + "max": None,
            prefix + "missing": 1.0,
            prefix + "since_last": None,
        }
    values = [value for _, value in samples]
    last_time = samples[-1][0]
    return {
        prefix + "latest": samples[-1][1],
        prefix + "mean": round(sum(values) / len(values), 6),
        prefix + "slope": _least_squares_slope(samples),
        prefix + "min": min(values),
        prefix + "max": max(values),
        prefix + "missing": 0.0,
        prefix + "since_last": round((index_time - last_time).total_seconds() / 3600, 4),
    }


def _least_squares_slope(samples: list[tuple[datetime, float]]) -> float | None:
    """Slope per hour with the first sample as origin; None when n < 3."""
    if len(samples) < 3:
        return None
    origin = samples[0][0]
    points = [((t - origin).total_seconds() / 3600, value) for t, value in samples]
    count = len(points)
    mean_t = sum(t for t, _ in points) / count
    mean_v = sum(v for _, v in points) / count
    denom = sum((t - mean_t) ** 2 for t, _ in points)
    if denom == 0:
        return None
    cov = sum((t - mean_t) * (v - mean_v) for t, v in points)
    return round(cov / denom, 6)
```

```python
def _parse_iso(value: str) -> datetime:
    parsed = datetime.fromisoformat(value)

def main() -> None:
    parser = argparse.ArgumentParser(description="Build landmark features next to a label CSV")
    parser.add_argument("--labels", required=True, help="CSV from derive_labels.py")
```

```python
def _validated_preprocess(
    raw: object, feature_order: list[str]
) -> dict[str, dict[str, float]] | None:
    if raw is None:
        return None
    if not isinstance(raw, dict):
        raise ValueError("model preprocess block is invalid")
    stats: dict[str, dict[str, float]] = {}
    for name in feature_order:
        entry = raw.get(name)
        if not isinstance(entry, dict):
            raise ValueError("model preprocess block is invalid")
        try:
            median = float(entry["median"])
            mean = float(entry["mean"])
            std = float(entry["std"])
        except (KeyError, TypeError, ValueError) as exc:
            raise ValueError("model preprocess block is invalid") from exc
        if not all(math.isfinite(value) for value in (median, mean, std)):
            raise ValueError("model preprocess block is invalid")
        stats[name] = {"median": median, "mean": mean, "std": std}
    return stats
```

```python
        payload = event.payload
        feature_order = self.feature_order
        missing_features = [
            feature for feature in feature_order if payload.get(feature) is None
        ]
        contributions: list[tuple[str, float, float | None]] = []
        linear = self.bias
        if self.preprocess is not None:
            for name in feature_order:
                stats = self.preprocess[name]
                raw = payload.get(name)
                # Plan Ruling R10: imputed features still count as missing.
                value = stats["median"] if raw is None else float(raw)
                std = stats["std"] if stats["std"] > 0 else 1.0
                contribution = self.weights.get(name, 0.0) * (value - stats["mean"]) / std
                linear += contribution
                contributions.append((name, contribution, value))
        else:
            for feature in feature_order:
                raw = payload.get(feature)
                if raw is None:
                    continue
                value = float(raw)
                contribution = self.weights.get(feature, 0) * value
                # Low blood pressure and oxygen saturation have negative raw values in the model.
                # Their clinical risk direction is represented by the signed contribution after centering.
                if feature == "systolic_bp":
                    contribution = self.weights[feature] * (value - 120)
                elif feature == "spo2":
                    contribution = self.weights[feature] * (value - 96)
                elif feature == "age":
                    contribution = self.weights[feature] * max(value - 50, 0)
                elif feature == "temperature":
                    contribution = self.weights[feature] * abs(value - 37.0)
                linear += contribution
                contributions.append((feature, contribution, value))
        score = round(self._sigmoid(linear), 4)
        band = (
            "critical"
            if score >= self.thresholds["critical"]
            else "high"
            if score >= self.thresholds["high"]
            else "medium"
            if score >= self.thresholds["medium"]
            else "low"

            tenant_id=event.tenant_id,
```

```python
def column_stats(rows: list[list[float]]) -> list[dict]:
    """Median/mean/std per raw column (population std; zero std guarded to 1)."""
    stats = []
    for column in zip(*rows):
        ordered = sorted(column)
        count = len(ordered)
        mean = sum(ordered) / count
        variance = sum((value - mean) ** 2 for value in ordered) / count
        median = (
            ordered[count // 2]
            if count % 2
            else (ordered[count // 2 - 1] + ordered[count // 2]) / 2
        )
        std = math.sqrt(variance) or 1.0
        stats.append({"median": median, "mean": mean, "std": std})
    return stats
```

```python
"""NEWS2 early-warning baseline (T5.4).

RCP NEWS2 Scale 1：五个生理组件按区间计 0–3 分；意识项（ACVPU）演示数据
缺测，不计入总分——报告与文档注明该局限（Plan Ruling R9）。
"""

from typing import Any, Mapping


def _points(value: float, bands: tuple[tuple[float, int], ...]) -> int:
    """bands: ascending (upper_bound_inclusive, points); entry (-1, p) = +inf."""
    for upper, points in bands:
        if upper < 0 or value <= upper:
            return points
    raise ValueError("value outside NEWS2 band table")


RR_BANDS: tuple[tuple[float, int], ...] = ((8, 3), (11, 1), (20, 0), (24, 2), (-1, 3))
SPO2_BANDS: tuple[tuple[float, int], ...] = ((91, 3), (93, 2), (95, 1), (-1, 0))
SBP_BANDS: tuple[tuple[float, int], ...] = ((90, 3), (100, 2), (110, 1), (219, 0), (-1, 3))
HR_BANDS: tuple[tuple[float, int], ...] = ((40, 3), (50, 1), (90, 0), (110, 1), (130, 2), (-1, 3))
TEMP_BANDS: tuple[tuple[float, int], ...] = ((35.0, 3), (36.0, 1), (38.0, 0), (39.0, 1), (-1, 2))

COMPONENTS: tuple[tuple[str, tuple[tuple[float, int], ...]], ...] = (
    ("respiratory_rate", RR_BANDS),
    ("spo2", SPO2_BANDS),
    ("systolic_bp", SBP_BANDS),
    ("heart_rate", HR_BANDS),
    ("temperature", TEMP_BANDS),
)


def news2_score(row: Mapping[str, Any]) -> tuple[int, str]:
    """Return (total score 0-15, band). Missing components are skipped."""
    total = 0
    highest_single = 0
    any_component = False
    for key, bands in COMPONENTS:
        raw = row.get(key)
        if raw is None or raw == "":
            continue
        points = _points(float(raw), bands)
        total += points
        highest_single = max(highest_single, points)
        any_component = True
    if not any_component:
        return 0, "low"
    # 官方升级规则：任一单参数 3 分 → high；否则按总分 0-4 low / 5-6 medium / >=7 high。
    if highest_single >= 3 or total >= 7:
        band = "high"
    elif total >= 5:
        band = "medium"
    else:
        band = "low"
    return total, band
```

```python
    raw = row.get("index_time") or row.get("occurred_at")
    if not raw:
        return datetime.now(UTC)
    parsed = datetime.fromisoformat(raw)

    payload = {
        feature: float(row[feature])
        for feature in FEATURES
        if row.get(feature, "") not in ("", None)
    }

def comparison_metrics(
    rows: Iterable[dict[str, str]], engine: RiskEngine, label_column: str
) -> dict[str, dict[str, float | None]]:
    """Three-way metrics on the same rows/labels: model (news2 median-imputed
    for v2 artifacts), news2 baseline (normalized /15, threshold 7/15),
    model_with_news2 (news2_score provided in the payload)."""
    from scripts.baselines import news2_score

    model_labels: list[int] = []
    model_scores: list[float] = []
    enriched_labels: list[int] = []
    enriched_scores: list[float] = []
    baseline_labels: list[int] = []
    baseline_scores: list[float] = []
    for index, row in enumerate(rows, start=1):
        try:
            label = int(row[label_column])
        except (KeyError, TypeError, ValueError) as exc:
            raise ValueError(f"row {index}: {label_column} must be 0 or 1") from exc
        if label not in (0, 1):

            engine.score(
                _make_event(row, _payload(row, include_news2=True), occurred_at, index)
            ).score
        )
        total, _ = news2_score(base_payload)
        baseline_labels.append(label)
        baseline_scores.append(total / 15)
    return {
        "model": binary_metrics(model_labels, model_scores, engine.thresholds["high"]),
        "news2": binary_metrics(baseline_labels, baseline_scores, 7 / 15),
```

```python
def _row_time(row: dict) -> datetime:
    raw = row.get("index_time") or row.get("occurred_at") or ""
    if not raw:
        raise ValueError("every row needs index_time or occurred_at for temporal splitting")
    parsed = datetime.fromisoformat(str(raw))
    return parsed if parsed.tzinfo else parsed.replace(tzinfo=UTC)


def rolling_time_split(rows: list[dict], n_splits: int) -> list[tuple[list[dict], list[dict]]]:
    """Expanding-window folds over time-ordered rows.

    Rows are sorted by time, cut into n_splits + 1 contiguous blocks; fold i
    trains on blocks 0..i-1 and tests on block i. Test blocks are disjoint and
    every train row precedes every test row (no temporal leakage).
    """
    if n_splits < 1:
        raise ValueError("n_splits must be >= 1")
    if len(rows) < n_splits + 1:
        raise ValueError(f"need at least {n_splits + 1} rows for {n_splits} splits")
    ordered = sorted(rows, key=_row_time)
    count = len(ordered) // (n_splits + 1)
    extra = len(ordered) % (n_splits + 1)
    blocks: list[list[dict]] = []
    start = 0
    for index in range(n_splits + 1):
        length = count + (1 if index < extra else 0)
        blocks.append(ordered[start : start + length])
        start += length
    folds: list[tuple[list[dict], list[dict]]] = []
    for index in range(1, n_splits + 1):
        train = [row for block in blocks[:index] for row in block]
        folds.append((train, blocks[index]))
    return folds


def bootstrap_metric(
    labels: list[int],
    scores: list[float],
    metric: Callable[[list[int], list[float]], float],
    n: int = 1000,
    *,
    seed: int | None = None,
) -> tuple[float, float, float]:
    """Percentile bootstrap: (point, lo, hi) at 2.5%/97.5% percentiles."""
    if not labels or len(labels) != len(scores):
        raise ValueError("labels and scores must be non-empty and the same length")
    rng = Random(seed)
    point = metric(labels, scores)
    size = len(labels)
    distribution = []
    for _ in range(n):
        indices = [rng.randrange(size) for _ in range(size)]
        distribution.append(metric([labels[i] for i in indices], [scores[i] for i in indices]))
    distribution.sort()
    lo = distribution[max(int(n * 0.025), 0)]
    hi = distribution[min(int(n * 0.975), n - 1)]
    return point, lo, hi


def main() -> None:
    parser = argparse.ArgumentParser(description="Rolling temporal validation with bootstrap CI")
    parser.add_argument("--input", type=Path, required=True)
    parser.add_argument("--label-column", default="label")
    parser.add_argument("--n-splits", type=int, default=3)
    parser.add_argument("--n-bootstrap", type=int, default=1000)
    parser.add_argument(
        "--target-id",
        choices=["deterioration_24h", "readmission_30d"],
        default="deterioration_24h",
    )
    parser.add_argument("--model", type=Path, default=None)
    args = parser.parse_args()
    model_path = args.model or Path(
        "data/model.json"
        if args.target_id == "deterioration_24h"
        else "data/readmission_model.json"
    )
    engine = RiskEngine(
        Settings(model_path=str(model_path), app_env="dev"), target_id=args.target_id
    )
    with args.input.open(newline="", encoding="utf-8-sig") as handle:
        rows = list(csv.DictReader(handle))
    folds = rolling_time_split(rows, args.n_splits)
    fold_reports = []
    oof_labels: list[int] = []
    oof_scores: list[float] = []
    for train, test in folds:
        labels, scores = evaluate_rows(test, engine, args.label_column)
        oof_labels.extend(labels)
        oof_scores.extend(scores)
        fold_reports.append(
            {
                "train_rows": len(train),
                "test_rows": len(test),
                "evaluation": binary_metrics(labels, scores, engine.thresholds["high"]),
            }
        )
    threshold = engine.thresholds["high"]

    def auprc(labels: list[int], scores: list[float]) -> float:
        return binary_metrics(labels, scores, threshold)["auprc"] or 0.0

    point, lo, hi = bootstrap_metric(
        oof_labels, oof_scores, auprc, n=args.n_bootstrap, seed=20260928
    )
    report = {
        "target_id": args.target_id,
        "model_version": engine.version,
        "folds": fold_reports,
        "out_of_fold": binary_metrics(oof_labels, oof_scores, threshold),
        "bootstrap_ci": {
            "metric": "auprc",
            "point": point,
            "lo": lo,
            "hi": hi,
            "n": args.n_bootstrap,
        },
    }
    print(json.dumps(report, ensure_ascii=False, indent=2))


if __name__ == "__main__":
    main()
```
