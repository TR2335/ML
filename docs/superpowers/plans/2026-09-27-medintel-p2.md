```python
    def _migrate_event_tenant_scope(self) -> None:
        """Re-key events to (tenant_id, event_id) so cross-tenant ids coexist."""
        table = self.connection.execute(
            "SELECT name FROM sqlite_master WHERE type = 'table' AND name = 'events'"
        ).fetchone()
        if not table:
            return
        columns = self.connection.execute("PRAGMA table_info(events)").fetchall()
        primary_key = [row["name"] for row in columns if row["pk"]]
        if primary_key == ["tenant_id", "event_id"]:
            return
        self.connection.execute("PRAGMA foreign_keys=OFF")
        try:
            with self.connection:
                self.connection.execute("ALTER TABLE events RENAME TO events_legacy")
                self.connection.executescript(
                    """
                    CREATE TABLE events (
                        event_id TEXT NOT NULL,
                        encounter_id TEXT NOT NULL,
                        patient_id TEXT NOT NULL,
                        tenant_id TEXT NOT NULL,
                        facility_id TEXT NOT NULL,
                        occurred_at TEXT NOT NULL,
                        data_json TEXT NOT NULL,
                        FOREIGN KEY (tenant_id, encounter_id)
                            REFERENCES patients(tenant_id, encounter_id),
                        PRIMARY KEY (tenant_id, event_id)
                    );
                    CREATE INDEX IF NOT EXISTS events_tenant_encounter_time
                        ON events(tenant_id, encounter_id, occurred_at);
                    INSERT INTO events(event_id, encounter_id, patient_id, tenant_id, facility_id, occurred_at, data_json)
                        SELECT event_id, encounter_id, patient_id, tenant_id, facility_id, occurred_at, data_json
                        FROM events_legacy;
                    DROP TABLE events_legacy;
                    """
                )
        finally:
            self.connection.execute("PRAGMA foreign_keys=ON")
```

注意：迁移对旧库先执行（检测旧 PK → 重建）；对新库无表时跳过，随后 `CREATE TABLE IF NOT EXISTS events` 直接建新结构——与既有两段迁移的协作方式一致。

```python
def test_concurrent_review_and_score_both_kept(tmp_path):
    from concurrent.futures import ThreadPoolExecutor

    store = Store(str(tmp_path / "store.db"))
    store.seed([{"patient_id": "pat-1", "encounter_id": "enc-1", "display_name": "测试患者"}], [])
    base = {"patient_id": "pat-1", "encounter_id": "enc-1", "model_version": "test"}
    # 初始分数：as_of 取旧时刻，保证并发 rescore 必然"更新"而非 stale
    store.upsert_score(
        RiskScore(
            **base,
            score=0.1,
            band="low",
            as_of=datetime(2026, 1, 1, 6, 0, tzinfo=UTC),
        )
    )

    def review_once() -> None:
        score = store.get_score("enc-1", "group-demo", "hospital-01")
        score.status = "resolved"
        score.human_review = "已处置"
        store.upsert_score(score, preserve_review=False)

    def rescore() -> str:
        return store.upsert_score(
            RiskScore(
                **base,
                score=0.9,
                band="critical",
                as_of=datetime(2026, 1, 1, 12, 0, tzinfo=UTC),
            )
        )

    with ThreadPoolExecutor(max_workers=4) as executor:
        futures = [executor.submit(fn) for fn in (review_once, rescore) * 10]
        for future in futures:
            future.result()

    # 全行写入下最终 score 取决于交错顺序（review 可能携带旧分数最后写入）——
    # 用一次确定性的最终 rescore 收敛分数字段；preserve_review 必须保住并发写入的复核状态。
    store.upsert_score(
        RiskScore(
            **base,
            score=0.9,
            band="critical",
            as_of=datetime(2026, 1, 1, 13, 0, tzinfo=UTC),
        )
    )
    final = store.get_score("enc-1", "group-demo", "hospital-01")
    assert final.status == "resolved"
    assert final.human_review == "已处置"
    assert final.score == 0.9
```

- [ ] **Step 4: 实现 SQLite BEGIN IMMEDIATE**

`app/store.py` `upsert_score` 重排事务边界（合并逻辑体不变）：

```python
    def upsert_score(self, score: RiskScore, *, preserve_review: bool = True) -> str:
        with self._lock:
            # BEGIN IMMEDIATE 升级为写事务，使跨进程的读-改-写原子化。
            self.connection.execute("BEGIN IMMEDIATE")
            try:
                previous = self.get_score(
                    score.encounter_id,
                    score.tenant_id,
                    score.facility_id,
                    target_id=score.target_id,
                )
                if previous:
                    new_as_of = score.as_of or score.scored_at
                    old_as_of = previous.as_of or previous.scored_at
                    if new_as_of < old_as_of:
                        self.connection.commit()
                        return "skipped_stale"
                    score.score_id = previous.score_id
                    if preserve_review and previous.status != "new":
                        score.status = previous.status
                        score.human_review = previous.human_review
                self.connection.execute(
                    "INSERT INTO scores(...) ...",  # 原语句逐字保留
                    (...),
                )
                self.connection.commit()
            except BaseException:
                self.connection.rollback()
                raise
        return "applied"
```

```python
def reconcile(store: Store, engines: dict[TargetId, RiskEngine]) -> int:
    """Backfill missing or stale scores from durable events without regressing newer ones."""
    count = 0
    for tenant_id, facility_id, encounter_id, latest in store.encounter_latest_event_times():
        scores = store.get_scores(encounter_id, tenant_id, facility_id)
        stale = False
        for target_id in engines:
            score = scores.get(target_id)
            if score is None:
                stale = True
                break
            effective = score.as_of or score.scored_at
            if effective < latest:
                stale = True
                break
        if not stale:
            continue
        events = store.events_for(encounter_id, tenant_id, facility_id)
        if not events:
            continue
        snapshot = store.feature_snapshot(encounter_id, tenant_id, facility_id, as_of=latest)
        scoring_event = max(events, key=lambda item: item.occurred_at).model_copy(
            update={"payload": snapshot}
        )
        for target_engine in engines.values():
            store.upsert_score(target_engine.score(scoring_event))
        count += 1
    return count
```

要点：`scoring_event` 取 `occurred_at` 最大者（保证 `as_of == latest`，满足测试断言且不制造回退）；`upsert_score` 的 stale 保护继续兜底。

```python
"""Shared ingestion validation helpers (API + import script)."""

from datetime import UTC, datetime, timedelta


def utc_now() -> datetime:
    return datetime.now(UTC)


def is_future_offset(
    occurred_at: datetime, *, max_skew_minutes: int, now: datetime | None = None
) -> bool:
    """True when the event time exceeds allowed future skew (naive treated as UTC)."""
    reference = now or utc_now()
    occurred = occurred_at if occurred_at.tzinfo else occurred_at.replace(tzinfo=UTC)
    return occurred > reference + timedelta(minutes=max_skew_minutes)


class ImportPartialFailure(RuntimeError):
    """Raised when a batch import fails mid-way; carries the partial summary."""

    def __init__(self, summary: dict, message: str):
        super().__init__(message)
        self.summary = summary
```

```python
    def import_records(self, patients: list[dict], events: list[ClinicalEvent]) -> dict:
        """Write a validated batch in one transaction; roll back completely on failure."""
        summary: dict = {
            "patients": 0,
            "accepted_events": 0,
            "duplicate_events": 0,
            "failed": 0,
            "last_ok": None,
        }
        with self._lock:
            self.connection.execute("BEGIN IMMEDIATE")
            try:
                for patient in patients:
                    tenant_id = patient.get("tenant_id", "group-demo")
                    facility_id = patient.get("facility_id", "hospital-01")
                    self.connection.execute(
                        "INSERT OR IGNORE INTO patients(encounter_id, patient_id, tenant_id, facility_id, data_json) VALUES (?, ?, ?, ?, ?)",
                        (
                            patient["encounter_id"],
                            patient["patient_id"],
                            tenant_id,
                            facility_id,
                            self._json(
                                {**patient, "tenant_id": tenant_id, "facility_id": facility_id}
                            ),
                        ),
                    )
                    summary["patients"] += 1
                    summary["last_ok"] = f"{tenant_id}/{patient['encounter_id']}"
                for event in events:
                    if self._insert_event(event):
                        summary["accepted_events"] += 1
                    else:
                        summary["duplicate_events"] += 1
                    summary["last_ok"] = f"{event.tenant_id}/{event.event_id}"
                self.connection.commit()
            except BaseException as exc:
                self.connection.rollback()
                summary["failed"] = 1
                raise ImportPartialFailure(summary, str(exc)) from exc
        return summary
```
